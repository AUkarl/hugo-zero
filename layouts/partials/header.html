{{- /* ------------------------------------------------------------ 导航项全部来自 config 的菜单（menus.zh.toml / menus.en.toml），支持： name 显示文字（可为空，配合 icon_only 做纯图标） url 站内路径或外链 weight 排序（数字越小越靠前） identifier / parent 父子关系 → 自动渲染成「点击展开的子菜单」 [main.params] external = true 外部链接（新标签页打开） icon = "fa-solid fa-xxx" Font Awesome 图标 image = "/img/x.svg" 自定义图片（与 icon 二选一） icon_only = true 只显示图标，文字转为无障碍标签 position = "controls" 放到右侧图标区（与主题/语言开关按 weight 统一排序） ------------------------------------------------------------ */ -}} {{- $themeWeight := .Site.Params.theme_switch_weight | default 90 -}} {{- $langWeight := .Site.Params.lang_switch_weight | default 100 -}} {{- $themeIcon := .Site.Params.theme_switch_icon | default "fas fa-sun" -}} {{- $langIcon := .Site.Params.lang_switch_icon | default "fas fa-globe" -}} {{- range .Site.Menus.main -}} {{- if and (not .Parent) (ne (.Params.position | default "nav") "controls") -}} {{- $href := .URL -}} {{- $external := .Params.external -}} {{- if and $href (not $external) -}}{{- $href = $href | relLangURL -}}{{- end -}} {{- $isCurrent := and $href (not $external) (or (eq (strings.TrimSuffix "/" $href) (strings.TrimSuffix "/" $.RelPermalink)) ($.HasMenuCurrent "main" .)) -}} {{- $iconOnly := .Params.icon_only | default false -}} {{- if .HasChildren -}}

{{- else -}} {{ with .Params.icon }}{{ end }}{{ with .Params.image }}{{ end }}{{ if not $iconOnly }}{{ .Name }}{{ end }} {{- end -}} {{- end -}} {{- end }}
{{- /* 右侧图标区：position="controls" 的菜单项 + 主题开关 + 语言开关，统一按 weight 排序 */ -}} {{- $ctrlItems := slice -}} {{- range .Site.Menus.main -}} {{- if and (not .Parent) (eq (.Params.position | default "") "controls") -}} {{- $ctrlItems = $ctrlItems | append (dict "kind" "menu" "weight" .Weight "entry" .) -}} {{- end -}} {{- end -}} {{- if .Site.Params.enable_theme_switch | default true -}} {{- $ctrlItems = $ctrlItems | append (dict "kind" "theme" "weight" $themeWeight) -}} {{- end -}} {{- if .Site.Params.enable_lang_switch | default true -}} {{- $ctrlItems = $ctrlItems | append (dict "kind" "lang" "weight" $langWeight) -}} {{- end -}} {{- range sort $ctrlItems "weight" -}} {{- if eq .kind "menu" -}} {{- $e := .entry -}} {{- $ehref := $e.URL -}} {{- if and $ehref (not $e.Params.external) -}}{{- $ehref = $ehref | relLangURL -}}{{- end -}} {{ with $e.Params.icon }}{{ end }}{{ with $e.Params.image }}{{ end }}{{ if not ($e.Params.icon_only | default false) }}{{ $e.Name }}{{ end }} {{- else if eq .kind "theme" -}} {{- else -}} {{- end -}} {{- end }}
